site = sys.argv[1]
out = {}
for name,(lat,lng) in pts.items():
    page=1
    while True:
        url=f"https://www.{site}/api/checkout/pub/pickup-points?geoCoordinates={lng};{lat}&page={page}&pageSize=100"
        req=urllib.request.Request(url,headers={'User-Agent':UA,'Accept':'application/json'})
        try:
            d=json.load(urllib.request.urlopen(req,timeout=40))
        except Exception as e:
            print(name,'ERR',e); break
        for it in d.get('items',[]):
            pp=it['pickupPoint']; out.setdefault(pp['id'],pp)
        print(name,page,d['paging'],len(out),flush=True)
        time.sleep(1.0)
        if page>=d['paging']['pages']: break
        page+=1
json.dump(out,open(f'pp_{site}.json','w',encoding='utf-8'),ensure_ascii=False,indent=1)
print('unique',len(out))
